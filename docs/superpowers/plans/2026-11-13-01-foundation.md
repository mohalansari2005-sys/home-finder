`backend/homefinder/db.py`:
```python
from collections.abc import Iterator
from contextlib import contextmanager

from sqlalchemy import Engine, create_engine
from sqlalchemy.orm import Session, sessionmaker

from homefinder.config import get_settings

_engine: Engine | None = None
_factory: sessionmaker[Session] | None = None


def get_engine() -> Engine:
    global _engine, _factory
    if _engine is None:
        _engine = create_engine(get_settings().database_url, pool_pre_ping=True, pool_size=5, max_overflow=5)
        _factory = sessionmaker(bind=_engine, expire_on_commit=False)
    return _engine


def SessionLocal() -> Session:
    get_engine()
    assert _factory is not None
    return _factory()


@contextmanager
def session_scope() -> Iterator[Session]:
    s = SessionLocal()
    try:
        yield s
        s.commit()
    except Exception:
        s.rollback()
        raise
    finally:
        s.close()
```

`backend/homefinder/textnorm.py`:
```python
import re

_LETTERS = str.maketrans({"أ": "ا", "إ": "ا", "آ": "ا", "ٱ": "ا", "ة": "ه", "ى": "ي", "ؤ": "و", "ئ": "ي"})
_DIACRITICS = re.compile(r"[ً-ْٰـ]")
_DIGITS = str.maketrans("٠١٢٣٤٥٦٧٨٩٫٬", "0123456789.,")


def to_western_digits(s: str) -> str:
    return s.translate(_DIGITS)


def normalize_ar(s: str | None) -> str:
    s = _DIACRITICS.sub("", s or "")
    s = s.translate(_LETTERS)
    return re.sub(r"\s+", " ", s).strip()


def normalize_district(s: str | None) -> str:
    return re.sub(r"^حي\s+", "", normalize_ar(s))
```

```python
from datetime import timedelta

from sqlalchemy import func, select, text, update
from sqlalchemy.dialects.postgresql import insert as pg_insert
from sqlalchemy.orm import Session

from homefinder.models import Job

MAX_ATTEMPTS = 3
BACKOFF = [timedelta(minutes=1), timedelta(minutes=10), timedelta(hours=1)]
STALE_AFTER = timedelta(minutes=15)


def enqueue(s: Session, type: str, payload: dict | None = None, *, dedupe_key: str | None = None,
            priority: int = 0, delay: timedelta | None = None) -> int | None:
    values = dict(type=type, payload=payload or {}, dedupe_key=dedupe_key, priority=priority)
    if delay:
        values["run_after"] = func.now() + delay
    stmt = pg_insert(Job).values(**values)
    if dedupe_key:
        stmt = stmt.on_conflict_do_nothing(index_elements=["dedupe_key"],
                                           index_where=text("status IN ('pending','claimed')"))
    return s.execute(stmt.returning(Job.id)).scalar_one_or_none()


def reclaim_stale(s: Session) -> int:
    cutoff = func.now() - STALE_AFTER
    failed = s.execute(update(Job).where(Job.status == "claimed", Job.claimed_at < cutoff, Job.attempts >= MAX_ATTEMPTS)
                       .values(status="failed", finished_at=func.now(), error="stale: worker never finished")).rowcount
    back = s.execute(update(Job).where(Job.status == "claimed", Job.claimed_at < cutoff)
                     .values(status="pending", claimed_at=None)).rowcount
    return failed + back


def claim(s: Session, types: list[str], n: int = 1) -> list[Job]:
    reclaim_stale(s)
    rows = list(s.scalars(
        select(Job).where(Job.status == "pending", Job.type.in_(types), Job.run_after <= func.now())
        .order_by(Job.priority.desc(), Job.id).limit(n).with_for_update(skip_locked=True)))
    for j in rows:
        j.status = "claimed"
        j.claimed_at = func.now()
        j.attempts += 1
    s.flush()
    for j in rows:
        s.refresh(j)
    return rows


def complete(s: Session, job_id: int, result: dict | None = None) -> bool:
    n = s.execute(update(Job).where(Job.id == job_id, Job.status.in_(["claimed", "pending"]))
                  .values(status="done", finished_at=func.now(), result=result, error=None)).rowcount
    return n == 1


def fail(s: Session, job_id: int, error: str) -> str:
    j = s.get(Job, job_id, with_for_update=True)
    if j is None or j.status == "done":
        return j.status if j else "missing"
    j.error = error[:4000]
    j.claimed_at = None
    if j.attempts >= MAX_ATTEMPTS:
        j.status = "failed"
        j.finished_at = func.now()
    else:
        j.status = "pending"
        j.run_after = func.now() + BACKOFF[max(j.attempts - 1, 0)]
    s.flush()
    return j.status
```

```python
import json
import logging
from collections.abc import Callable
from typing import Protocol

log = logging.getLogger(__name__)

GRAPHQL_URL = "https://sa.aqar.fm/graphql"
HOME_URL = "https://sa.aqar.fm/"
API_HEADERS = {"accept": "*/*", "content-type": "application/json", "origin": "https://sa.aqar.fm",
               "referer": HOME_URL, "req-app": "web", "app-version": "0.20.33", "accept-language": "ar,en;q=0.9"}


class Blocked(Exception):
    pass


class Transport(Protocol):
    def post_graphql(self, body: dict) -> dict: ...
    def close(self) -> None: ...


class CurlCffiTransport:
    def __init__(self, impersonate: str = "chrome"):
        from curl_cffi import requests as cr
        self._s = cr.Session(impersonate=impersonate)
        self._warmed = False

    def post_graphql(self, body: dict) -> dict:
        if not self._warmed:
            self._s.get(HOME_URL, timeout=30)
            self._warmed = True
        r = self._s.post(GRAPHQL_URL, json=body, headers=API_HEADERS, timeout=30)
        if r.status_code != 200 or "json" not in r.headers.get("content-type", ""):
            raise Blocked(f"curl_cffi: HTTP {r.status_code}")
        return r.json()

    def close(self) -> None:
        self._s.close()


_FETCH_JS = """async ([url, body, headers]) => {
  const r = await fetch(url, {method: 'POST', headers, body: JSON.stringify(body), credentials: 'include'});
  return {status: r.status, ct: r.headers.get('content-type') || '', text: await r.text()};
}"""


class PlaywrightTransport:
    def __init__(self, headless: bool = True):
        self._headless = headless
        self._pw = self._browser = self._page = None

    def _ensure(self):
        if self._page is not None:
            return
        from playwright.sync_api import sync_playwright
        self._pw = sync_playwright().start()
        self._browser = self._pw.chromium.launch(headless=self._headless)
        self._page = self._browser.new_page(locale="ar-SA")
        self._page.goto(HOME_URL, wait_until="domcontentloaded", timeout=60_000)
        self._page.wait_for_timeout(3000)

    def post_graphql(self, body: dict) -> dict:
        self._ensure()
        headers = {k: v for k, v in API_HEADERS.items() if k in ("content-type", "req-app", "app-version")}
        res = self._page.evaluate(_FETCH_JS, [GRAPHQL_URL, body, headers])
        if res["status"] != 200 or "json" not in res["ct"]:
            raise Blocked(f"playwright: HTTP {res['status']}")
        return json.loads(res["text"])

    def close(self) -> None:
        if self._browser:
            self._browser.close()
        if self._pw:
            self._pw.stop()
        self._pw = self._browser = self._page = None


class FallbackTransport:
    """Tries transports in order; once one is Blocked, moves to the next for the rest of the run."""

    def __init__(self, factories: list[Callable[[], Transport]]):
        self._factories = factories
        self._i = 0
        self._t: Transport | None = None

    def post_graphql(self, body: dict) -> dict:
        while True:
            if self._t is None:
                self._t = self._factories[self._i]()
            try:
                return self._t.post_graphql(body)
            except Blocked as e:
                log.warning("aqar transport %d blocked: %s", self._i, e)
                self._t.close()
                self._t = None
                if self._i + 1 >= len(self._factories):
                    raise
                self._i += 1

    def close(self) -> None:
        if self._t:
            self._t.close()
            self._t = None
```

```python
"""Scrub personal data from a captured JSON response before committing it to the PUBLIC repo.
Usage: python scripts/scrub_fixture.py raw.json clean.json"""
import json
import re
import sys

FAKE_PHONE = "0500000000"
PHONE_RE = re.compile(r"(?:\+?966|0)5\d{8}")
PHONE_KEYS = {"phone", "mobile", "whatsapp", "phone_number"}
ZERO_KEYS = {"user_id", "employee_user_id", "mgr_user_id", "rega_id"}
NULL_KEYS = {"img", "avatar", "bml_url"}
REDACT_KEYS = {"deed_number": "0000000000", "ad_license_number": "0000000000"}


def scrub(obj, parent_key: str | None = None):
    if isinstance(obj, dict):
        out = {}
        for k, v in obj.items():
            if k in PHONE_KEYS and v:
                out[k] = FAKE_PHONE
            elif k == "name" and parent_key in ("user", "author", "seller"):
                out[k] = "مالك تجريبي"
            elif k in ZERO_KEYS and v is not None:
                out[k] = 0
            elif k in NULL_KEYS:
                out[k] = None
            elif k in REDACT_KEYS and v:
                out[k] = REDACT_KEYS[k]
            else:
                out[k] = scrub(v, k)
        return out
    if isinstance(obj, list):
        return [scrub(v, parent_key) for v in obj]
    if isinstance(obj, str):
        return PHONE_RE.sub(FAKE_PHONE, obj)
    return obj


if __name__ == "__main__":
    src, dst = sys.argv[1], sys.argv[2]
    with open(src, encoding="utf-8") as f:
        data = json.load(f)
    with open(dst, "w", encoding="utf-8") as f:
        json.dump(scrub(data), f, ensure_ascii=False, indent=1)
```

```python
import logging
import re

from homefinder.textnorm import to_western_digits

log = logging.getLogger(__name__)

PRICE = (100_000, 100_000_000)
BUILT = (50, 5_000)
LAND = (50, 20_000)
AGE = (0, 80)
STREET = (4, 100)
ROOMS = (1, 20)
_NUM = re.compile(r"-?\d+(?:\.\d+)?")


def parse_number(v) -> float | None:
    if v is None or isinstance(v, bool):
        return None
    if isinstance(v, (int, float)):
        return float(v)
    m = _NUM.search(to_western_digits(str(v)).replace(",", ""))
    return float(m.group()) if m else None


def clean_float(v, lo: float, hi: float) -> float | None:
    n = parse_number(v)
    if n is None:
        return None
    if not lo <= n <= hi:
        log.warning("value %r outside sanity range [%s, %s]", v, lo, hi)
        return None
    return n


def clean_int(v, lo: float, hi: float) -> int | None:
    n = clean_float(v, lo, hi)
    return int(round(n)) if n is not None else None
```

```python
import random
import time
from collections.abc import Callable, Iterator
from typing import Protocol

from homefinder.schemas import HealthReport, RawListing, SearchQuery


class SourceAdapter(Protocol):
    name: str
    def search(self, q: SearchQuery) -> Iterator[RawListing]: ...
    def fetch(self, source_id: str) -> RawListing | None: ...
    def health(self, sample: list[RawListing]) -> HealthReport: ...


class Pacer:
    """Polite delay between requests: uniform(2,3)s, or uniform(5,10)s with 1% probability. First call is free."""

    def __init__(self, sleep: Callable[[float], None] = time.sleep, rng: random.Random | None = None):
        self._sleep = sleep
        self._rng = rng or random.Random()
        self._first = True

    def wait(self) -> None:
        if self._first:
            self._first = False
            return
        self._sleep(self._rng.uniform(5, 10) if self._rng.random() < 0.01 else self._rng.uniform(2, 3))


def default_health(sample: list[RawListing], min_sample: int = 10, threshold: float = 0.5) -> HealthReport:
    n = len(sample)
    if n == 0:
        return HealthReport(ok=True, sample_size=0, missing_ratio={})
    ratio = {
        "price": sum(r.fields.price is None for r in sample) / n,
        "location": sum(r.fields.lat is None and r.fields.district is None for r in sample) / n,
        "area": sum(r.fields.built_area_m2 is None and r.fields.land_area_m2 is None for r in sample) / n,
    }
    bad = [k for k, v in ratio.items() if v > threshold]
    if n >= min_sample and bad:
        return HealthReport(ok=False, sample_size=n, missing_ratio=ratio,
                            reason=f"missing {', '.join(bad)} in >{int(threshold * 100)}% of {n} results")
    return HealthReport(ok=True, sample_size=n, missing_ratio=ratio)
```

```python
class AqarAdapter:
    name = "aqar"

    def __init__(self, transport: Transport, pacer: Pacer | None = None):
        self._t = transport
        self._pacer = pacer or Pacer()
        self._district_cache: dict[tuple[int, int], dict[str, int]] = {}

    def _post(self, body: dict) -> dict:
        self._pacer.wait()
        resp = self._t.post_graphql(body)
        if resp.get("errors") and not resp.get("data"):
            raise RuntimeError(f"aqar graphql error: {resp['errors']}")
        return resp

    def _district_ids(self, category: int, city_id: int, names: list[str]) -> list[int]:
        key = (category, city_id)
        if key not in self._district_cache:
            data = self._post(districts_body(category, city_id))
            self._district_cache[key] = {normalize_district(d["name"]): d["district_id"]
                                         for d in data["data"]["Web"]["districts"] or []}
        table = self._district_cache[key]
        ids = []
        for n in names:
            did = table.get(normalize_district(n))
            if did is None:
                log.warning("aqar: unknown district %r", n)
            else:
                ids.append(did)
        return ids

    def search(self, q: SearchQuery) -> Iterator[RawListing]:
        city_id = CITY_IDS[q.city]
        for ptype in q.property_types:
            cat = CATEGORY_BY_TYPE[ptype]
            district_ids: list[int | None] = self._district_ids(cat, city_id, q.districts) if q.districts else [None]
            for did in district_ids:
                for page in range(q.max_pages):
                    where = {"category": {"eq": cat}, "city_id": {"eq": city_id}, "published": {"eq": 1}}
                    if did is not None:
                        where["district_id"] = {"eq": did}
                    rows = self._post(find_body(page * PAGE, where))["data"]["Web"]["find"]["listings"] or []
                    for d in rows:
                        r = parse_listing(d)
                        if q.price_max and r.fields.price and r.fields.price > q.price_max:
                            continue
                        yield r
                    if len(rows) < PAGE:
                        break

    def fetch(self, source_id: str) -> RawListing | None:
        rows = self._post(find_body(0, {"id": {"eq": int(source_id)}}, size=1))["data"]["Web"]["find"]["listings"] or []
        return parse_listing(rows[0]) if rows and str(rows[0]["id"]) == source_id else None

    def health(self, sample: list[RawListing]) -> HealthReport:
```

```python
from dataclasses import dataclass

from geoalchemy2.elements import WKTElement
from sqlalchemy import func, select
from sqlalchemy.orm import Session

from homefinder import jobs
from homefinder.models import Listing, ListingSource, PriceHistory
from homefinder.schemas import RawListing

LISTING_FIELDS = [f for f in RawListing.model_fields["fields"].annotation.model_fields if f not in ("lat", "lng")]


@dataclass
class IngestResult:
    listing_id: int
    created: bool
    price_changed: bool


def set_location(listing: Listing, lat: float | None, lng: float | None, precision: str) -> None:
    if lat is None or lng is None:
        listing.lat = listing.lng = listing.geom = None
        listing.geo_precision = "none"
        return
    listing.lat, listing.lng = lat, lng
    listing.geom = WKTElement(f"POINT({lng} {lat})", srid=4326)
    listing.geo_precision = precision


def _canonical(s: Session, listing: Listing) -> Listing:
    seen = set()
    while listing.merged_into_id and listing.id not in seen:
        seen.add(listing.id)
        listing = s.get(Listing, listing.merged_into_id)
    return listing


def ingest_raw(s: Session, raw: RawListing) -> IngestResult:
    f = raw.fields
    src = s.scalar(select(ListingSource).where(ListingSource.source == raw.source,
                                               ListingSource.source_id == raw.source_id))
    if src is None:
        listing = Listing(**{k: getattr(f, k) for k in LISTING_FIELDS})
        set_location(listing, f.lat, f.lng, raw.geo_precision or "none")
```

```python
import logging
from collections.abc import Callable
from contextlib import AbstractContextManager
from dataclasses import dataclass
from typing import Any

from sqlalchemy.orm import Session

from homefinder import adapters_state
from homefinder.schemas import SearchQuery
from homefinder.worker.ingest import ingest_raw

log = logging.getLogger(__name__)


@dataclass
class WorkerDeps:
    session_factory: Callable[[], AbstractContextManager[Session]]
    adapters: dict[str, Callable[[], Any]]
    llm: Any
    google: Any
    photo_sink: Any
    http: Any


def handle_scrape_search(deps: WorkerDeps, payload: dict) -> dict:
    source = payload["source"]
    with deps.session_factory() as s:
        if adapters_state.is_paused(s, source):
            return {"skipped": "paused"}
    adapter = deps.adapters[source]()
    try:
        raws = list(adapter.search(SearchQuery(**payload.get("query", {}))))
    finally:
        close = getattr(getattr(adapter, "_t", None), "close", None)
        if close:
            close()
    report = adapter.health(raws)
    if not report.ok:
        with deps.session_factory() as s:
            adapters_state.pause(s, source, report.reason or "unhealthy")
        return {"skipped": "unhealthy", "reason": report.reason, "fetched": len(raws)}
    new = 0
    for raw in raws:
        with deps.session_factory() as s:
            new += ingest_raw(s, raw).created
    with deps.session_factory() as s:
        adapters_state.mark_ok(s, source)
    return {"fetched": len(raws), "ingested": len(raws), "new": new}


HANDLERS: dict[str, Callable[[WorkerDeps, dict], dict]] = {
    "scrape_search": handle_scrape_search,
}


def dispatch(deps: WorkerDeps, job_type: str, payload: dict) -> dict:
    handler = HANDLERS.get(job_type)
    if handler is None:
        raise ValueError(f"unknown job type {job_type}")
    return handler(deps, payload)
```

```python
class BudgetExceeded(Exception):
    pass


def month_key(now: datetime | None = None) -> str:
    return (now or datetime.now(UTC)).astimezone(RIYADH).strftime("%Y-%m")


def _cap(provider: Provider) -> float:
    st = get_settings()
    return st.openai_monthly_cap_usd if provider == "openai" else st.google_monthly_cap_usd


def spend(provider: Provider, cost_usd: float) -> None:
    with session_scope() as s:
        row = s.scalar(select(ApiUsage).where(ApiUsage.provider == provider, ApiUsage.month == month_key())
                       .with_for_update())
        if row is None:
            row = ApiUsage(provider=provider, month=month_key(), cost_usd=0.0, calls=0)
            s.add(row)
        if row.cost_usd >= _cap(provider):
            raise BudgetExceeded(f"{provider} monthly cap {_cap(provider)} USD reached")
        row.cost_usd += cost_usd
        row.calls += 1
```

```python
from collections.abc import Callable
from datetime import datetime

import httpx

from homefinder import usage
from homefinder.config import get_settings

GEOCODE_URL = "https://maps.googleapis.com/maps/api/geocode/json"
MATRIX_URL = "https://routes.googleapis.com/distanceMatrix/v2:computeRouteMatrix"
COST = {"geocode": 0.005, "matrix_element": 0.01}


def _wp(p: tuple[float, float]) -> dict:
    return {"waypoint": {"location": {"latLng": {"latitude": p[0], "longitude": p[1]}}}}


class GoogleClient:
    def __init__(self, api_key: str, http: httpx.Client, spend: Callable[[float], None] | None = None):
        self._key, self._http = api_key, http
        self._spend = spend or (lambda c: usage.spend("google", c))

    def geocode(self, address: str) -> tuple[float, float] | None:
        self._spend(COST["geocode"])
        r = self._http.get(GEOCODE_URL, params={"address": address, "key": self._key, "language": "ar", "region": "sa"}, timeout=20)
        r.raise_for_status()
        data = r.json()
        if data.get("status") != "OK" or not data.get("results"):
            return None
        loc = data["results"][0]["geometry"]["location"]
        return loc["lat"], loc["lng"]

    def drive_minutes(self, origin: tuple[float, float], destinations: list[tuple[float, float]],
                      departure: datetime) -> list[int | None]:
        self._spend(COST["matrix_element"] * len(destinations))
        body = {"origins": [_wp(origin)], "destinations": [_wp(d) for d in destinations], "travelMode": "DRIVE",
                "routingPreference": "TRAFFIC_AWARE", "departureTime": departure.isoformat().replace("+00:00", "Z")}
        r = self._http.post(MATRIX_URL, json=body, timeout=30, headers={
            "X-Goog-Api-Key": self._key, "X-Goog-FieldMask": "originIndex,destinationIndex,duration,condition"})
        r.raise_for_status()
        out: list[int | None] = [None] * len(destinations)
        for el in r.json():
            if el.get("condition") == "ROUTE_EXISTS" and el.get("duration"):
                out[el["destinationIndex"]] = round(int(el["duration"].rstrip("s")) / 60)
        return out


def make_google(http: httpx.Client) -> GoogleClient | None:
    key = get_settings().google_api_key
    return GoogleClient(key, http) if key else None
```

```python
from typing import Protocol

from homefinder import usage
from homefinder.config import get_settings


class LLMClient(Protocol):
    def extract_json(self, *, system: str, user: str, schema: dict, name: str) -> dict: ...


class OpenAIClient:
    def __init__(self, api_key: str, model_small: str, model_large: str, price_in: float, price_out: float,
                 spend: Callable[[float], None] | None = None, client=None):
        if client is None:
            from openai import OpenAI
            client = OpenAI(api_key=api_key)
        self._c, self.model_small, self.model_large = client, model_small, model_large
        self._pin, self._pout = price_in, price_out
        self._spend = spend or (lambda c: usage.spend("openai", c))

    def extract_json(self, *, system: str, user: str, schema: dict, name: str) -> dict:
        self._spend(0.0)  # budget check
        resp = self._c.chat.completions.create(
            model=self.model_small,
            messages=[{"role": "system", "content": system}, {"role": "user", "content": user}],
            response_format={"type": "json_schema", "json_schema": {"name": name, "schema": schema, "strict": True}},
        )
        u = resp.usage
        self._spend(u.prompt_tokens * self._pin / 1e6 + u.completion_tokens * self._pout / 1e6)
        return json.loads(resp.choices[0].message.content)


def make_llm() -> OpenAIClient | None:
    st = get_settings()
    if not st.openai_api_key:
        return None
    return OpenAIClient(st.openai_api_key, st.openai_model_small, st.openai_model_large,
```

```python
from datetime import UTC, datetime, time, timedelta
from zoneinfo import ZoneInfo

from sqlalchemy import select
from sqlalchemy.orm import Session

from homefinder.models import CommuteCache, Listing, Place

RIYADH = ZoneInfo("Asia/Riyadh")
WORKDAYS = {6, 0, 1, 2, 3}  # Python weekday(): Sun=6, Mon=0 .. Thu=3
DEPARTURE = {"school": time(7, 30), "work": time(7, 30), "university": time(7, 30), "airport": time(18, 0)}


def next_workday_at(t: time, now: datetime) -> datetime:
    local = now.astimezone(RIYADH)
    for add in range(0, 8):
        day = (local + timedelta(days=add)).date()
        cand = datetime.combine(day, t, tzinfo=RIYADH)
        if day.weekday() in WORKDAYS and cand > local + timedelta(minutes=10):
            return cand.astimezone(UTC)
    raise AssertionError("unreachable")


def origin_key(lat: float, lng: float) -> str:
    return f"{lat:.3f},{lng:.3f}"


def compute_commutes(s: Session, google, listing: Listing, now: datetime | None = None) -> dict[str, int | None]:
    if listing.lat is None or listing.lng is None:
        return {}
    now = now or datetime.now(UTC)
    key = origin_key(listing.lat, listing.lng)
    places = list(s.scalars(select(Place)))
    cached = {c.place_id: c.minutes for c in s.scalars(select(CommuteCache).where(CommuteCache.origin_key == key))}
    result = {p.kind: cached[p.id] for p in places if p.id in cached}
    missing = [p for p in places if p.id not in cached]
    by_dep: dict[time, list[Place]] = {}
    for p in missing:
        by_dep.setdefault(DEPARTURE.get(p.kind, time(7, 30)), []).append(p)
    origin = (round(listing.lat, 3), round(listing.lng, 3))
    for dep, group in by_dep.items():
        mins = google.drive_minutes(origin, [(p.lat, p.lng) for p in group], next_workday_at(dep, now))
        for p, m in zip(group, mins):
            s.merge(CommuteCache(origin_key=key, place_id=p.id, minutes=m))
            result[p.kind] = m
    return result
```

```python
import logging
from io import BytesIO
from typing import Protocol

import httpx

log = logging.getLogger(__name__)
MAX_PHOTOS = 10
MAX_SIDE = 1600


class PhotoSink(Protocol):
    def upload(self, listing_id: int, position: int, data: bytes, phash: str, source_url: str) -> None: ...


def process_image(data: bytes) -> tuple[bytes, str]:
    import imagehash
    from PIL import Image, ImageOps
    im = ImageOps.exif_transpose(Image.open(BytesIO(data))).convert("RGB")
    im.thumbnail((MAX_SIDE, MAX_SIDE))
    out = BytesIO()
    im.save(out, "WEBP", quality=80)
    return out.getvalue(), str(imagehash.phash(im))


def fetch_photos(http: httpx.Client, sink: PhotoSink, listing_id: int, urls: list[str]) -> int:
    pos = 0
    for url in urls[:MAX_PHOTOS]:
        try:
            r = http.get(url, timeout=30, follow_redirects=True)
            r.raise_for_status()
            data, ph = process_image(r.content)
            sink.upload(listing_id, pos, data, ph, url)
            pos += 1
        except Exception as e:  # noqa: BLE001 — one bad photo must not stop the rest
            log.warning("photo %s failed: %s", url, e)
    return pos
```

```python
def build_schema(fields: list[str]) -> dict:
    return {"type": "object", "properties": {f: EXTRACTABLE[f] for f in fields},
            "required": fields, "additionalProperties": False}


def extract_missing(llm, listing: Listing) -> dict:
    text = "\n".join(x for x in (listing.title, listing.description) if x)
    missing = [f for f in EXTRACTABLE if getattr(listing, f) is None]
    if not text.strip() or not missing:
        return {}
    raw = llm.extract_json(system=SYSTEM, user=text[:6000], schema=build_schema(missing), name="listing_fields")
    out = {}
    for f in missing:
        v = raw.get(f)
        if v is None:
            continue
        v = _CLEAN[f](v) if f in _CLEAN else v
        if v is not None:
            out[f] = v
    return out
```

```python
import logging
from datetime import timedelta

from sqlalchemy import func, select

from homefinder import jobs
from homefinder.models import Listing, ListingSource
from homefinder.usage import BudgetExceeded
from homefinder.worker.enrich.commute import compute_commutes
from homefinder.worker.enrich.extract import extract_missing
from homefinder.worker.enrich.geo import ensure_location
from homefinder.worker.enrich.photos import fetch_photos

log = logging.getLogger(__name__)
RETRY_DELAY = timedelta(hours=6)


def enrich_listing(deps, listing_id: int) -> dict:
    requeue = False
    with deps.session_factory() as s:
        l = s.get(Listing, listing_id)
        if l is None or l.merged_into_id:
            return {"skipped": True}
        steps = []
        if deps.llm is not None:
            steps.append(("extract", lambda: [setattr(l, k, v) for k, v in extract_missing(deps.llm, l).items()]))
        steps.append(("geo", lambda: ensure_location(deps.google, l)))
        for name, step in steps:
            try:
                step()
            except BudgetExceeded as e:
                log.warning("enrich %s: %s skipped: %s", listing_id, name, e)
                requeue = True
            except Exception as e:  # noqa: BLE001 — provider outage: keep partial data, retry later
                log.warning("enrich %s: %s failed: %s", listing_id, name, e)
                requeue = True
        if l.lat is not None and deps.google is not None:
            try:
                l.commutes = compute_commutes(s, deps.google, l)
            except Exception as e:  # noqa: BLE001
                log.warning("enrich %s: commute failed: %s", listing_id, e)
                requeue = True
        l.enriched_at = func.now()
        need_photos = not l.photos_fetched and deps.photo_sink is not None and deps.http is not None
        urls = next((src.photo_urls for src in s.scalars(select(ListingSource).where(
            ListingSource.listing_id == listing_id)) if src.photo_urls), [])
        commutes = len(l.commutes or {})
        if requeue:
            jobs.enqueue(s, "enrich_listing", {"listing_id": listing_id},
                         dedupe_key=f"enrich-retry:{listing_id}", delay=RETRY_DELAY)
    photos = 0
    if need_photos and urls:
        photos = fetch_photos(deps.http, deps.photo_sink, listing_id, urls)
        with deps.session_factory() as s:
            s.get(Listing, listing_id).photos_fetched = True
    return {"commutes": commutes, "photos": photos, "requeued": requeue}
```

```python
_attempts: dict[str, deque] = defaultdict(deque)


def reset_rate_limit() -> None:
    _attempts.clear()


def _ip(request: Request) -> str:
    return request.headers.get("x-real-ip") or (request.client.host if request.client else "?")


class LoginBody(BaseModel):
    password: str


@router.post("/login", status_code=204)
def login(body: LoginBody, request: Request):
    q = _attempts[_ip(request)]
    now = time.monotonic()
    while q and now - q[0] > WINDOW:
        q.popleft()
    if len(q) >= MAX_ATTEMPTS:
        raise HTTPException(429, "too many attempts")
    h = get_settings().family_password_hash.encode()
    if not h or not bcrypt.checkpw(body.password.encode(), h):
        q.append(now)
        raise HTTPException(401, "wrong password")
    request.session["family"] = True
    return Response(status_code=204)
```

```python
import httpx


class WorkerClient:
    def __init__(self, base_url: str, token: str, http: httpx.Client | None = None):
        self._h = http or httpx.Client(base_url=base_url, timeout=60, headers={"Authorization": f"Bearer {token}"})

    def claim(self, types: list[str], n: int = 1) -> list[dict]:
        r = self._h.post("/worker/claim", json={"types": types, "n": n}); r.raise_for_status(); return r.json()

    def complete(self, job_id: int, result: dict | None) -> None:
        self._h.post(f"/worker/complete/{job_id}", json={"result": result}).raise_for_status()

    def fail(self, job_id: int, error: str) -> None:
        self._h.post(f"/worker/fail/{job_id}", json={"error": error[:4000]}).raise_for_status()

    def heartbeat(self, hostname: str, version: str, current_job_id: int | None) -> None:
        self._h.post("/worker/heartbeat", json={"hostname": hostname, "version": version,
                                                "current_job_id": current_job_id}).raise_for_status()

    def upload(self, listing_id: int, position: int, data: bytes, phash: str, source_url: str) -> None:
        r = self._h.post("/worker/photos", data={"listing_id": listing_id, "position": position, "phash": phash,
                                                 "source_url": source_url},
                         files={"file": (f"{position}.webp", data, "image/webp")})
```

```python
import logging
import socket
import time
from collections.abc import Callable

import httpx

from homefinder import __version__
from homefinder.config import get_settings
from homefinder.db import session_scope
from homefinder.integrations.google import make_google
from homefinder.integrations.llm import make_llm
from homefinder.worker import scheduler
from homefinder.worker.client import WorkerClient
from homefinder.worker.handlers import HANDLERS, WorkerDeps, dispatch
from homefinder.worker.sources.aqar import AqarAdapter, make_default_transport

log = logging.getLogger("homefinder.worker")
HEARTBEAT_EVERY = 60.0
OFFLINE_BACKOFF = 30.0


def _schedule(interval_hours: float) -> None:
    with session_scope() as s:
        scheduler.maybe_schedule(s, interval_hours)


def run_loop(deps, client, *, stop: Callable[[], bool], sleep=time.sleep, poll_interval: float = 5.0,
             interval_hours: float = 4.0, hostname: str = "mac") -> None:
    last_beat = -1e9
    while not stop():
        try:
            if time.monotonic() - last_beat > HEARTBEAT_EVERY:
                client.heartbeat(hostname, __version__, None)
                last_beat = time.monotonic()
            _schedule(interval_hours)
            claimed = client.claim(list(HANDLERS), 1)
        except Exception as e:  # noqa: BLE001 — server/DB unreachable (Mac off Tailscale, server restarting)
            log.warning("server unreachable: %s", e)
            sleep(OFFLINE_BACKOFF)
            continue
        if not claimed:
            sleep(poll_interval)
            continue
        job = claimed[0]
        log.info("job %s %s", job["id"], job["type"])
        try:
            result = dispatch(deps, job["type"], job["payload"])
            client.complete(job["id"], result)
        except Exception as e:  # noqa: BLE001
            log.exception("job %s failed", job["id"])
            try:
                client.fail(job["id"], f"{type(e).__name__}: {e}")
            except Exception:  # noqa: BLE001
                log.warning("could not report failure; job will be reclaimed after 15 min")


def build_deps(client: WorkerClient) -> WorkerDeps:
    http = httpx.Client(timeout=30, headers={"user-agent": "Mozilla/5.0 (Macintosh) HomeFinder/0.1"})
    return WorkerDeps(session_factory=session_scope,
                      adapters={"aqar": lambda: AqarAdapter(make_default_transport())},
                      llm=make_llm(), google=make_google(http), photo_sink=client, http=http)


def main() -> None:
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(name)s %(message)s")
    st = get_settings()
    client = WorkerClient(st.server_url, st.worker_token)
    run_loop(build_deps(client), client, stop=lambda: False, interval_hours=st.scrape_interval_hours,
             hostname=socket.gethostname())


if __name__ == "__main__":
    main()
```

```ts
export function useBreakpoint(): Breakpoint {
  const [bp, setBp] = useState(() => getBreakpoint(window.innerWidth))

    window.addEventListener('resize', on)
```

```ts
export class ApiError extends Error {
  constructor(public status: number, message: string) { super(message) }
}

export async function api<T>(path: string, init: RequestInit = {}): Promise<T> {
  const r = await fetch(path, {
    credentials: 'include',
    ...init,
    headers: { 'content-type': 'application/json', ...(init.headers ?? {}) },
  })
  if (r.status === 401 && !path.startsWith('/api/login') && !path.startsWith('/api/me')) {
    if (!location.pathname.startsWith('/login')) location.assign('/login')
  }
  if (!r.ok) throw new ApiError(r.status, await r.text())
  if (r.status === 204) return undefined as T
  return r.json() as Promise<T>
}
```

```tsx
      ))}

  )
```

```tsx
      <div className={`h-full overflow-y-auto ${bp === 'desktop' ? 'w-[440px]' : 'w-1/2'} shrink-0 border-e border-slate-200`}>{list(1)}</div>
```

```bash
#!/usr/bin/env bash
```

```bash
BACKEND="$ROOT/backend"

PLIST="$HOME/Library/LaunchAgents/com.homefinder.worker.plist"

[ -f "$BACKEND/.env" ] || { echo "Create $BACKEND/.env (DATABASE_URL to server Tailscale IP, SERVER_URL, WORKER_TOKEN, API keys)"; exit 1; }
cd "$BACKEND"
[ -d .venv ] || python3.12 -m venv .venv
.venv/bin/pip install -q -e ".[worker]"
.venv/bin/playwright install chromium
mkdir -p "$LOGS" "$(dirname "$PLIST")"
sed -e "s#__BACKEND__#$BACKEND#g" -e "s#__LOGS__#$LOGS#g" "$ROOT/deploy/worker/com.homefinder.worker.plist.template" > "$PLIST"
launchctl bootout "gui/$(id -u)/com.homefinder.worker" 2>/dev/null || true
launchctl bootstrap "gui/$(id -u)" "$PLIST"
echo "worker installed; logs: $LOGS"
```
